cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(scenes VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
file(GLOB SCENES_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
add_library(scenes STATIC ${SCENES_SOURCES})
add_library(chopfractal::scenes ALIAS scenes)
# Position independent so the module can be linked into a plugin (a shared module) in any project.
set_target_properties(scenes PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(scenes PUBLIC cxx_std_17)
target_link_libraries(scenes PUBLIC chopfractal::chop_contracts)
target_include_directories(scenes PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
if(MSVC)
  target_compile_options(scenes PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(scenes PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB SCENES_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_scenes ${SCENES_TESTS})
  target_link_libraries(test_scenes PRIVATE scenes)
  add_executable(example_scenes examples/minimal.cpp)
  target_link_libraries(example_scenes PRIVATE scenes)
  if(MSVC)
    target_compile_options(test_scenes PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_scenes PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_scenes PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_scenes PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME scenes.unit COMMAND test_scenes)
  add_test(NAME scenes.example COMMAND example_scenes)
endif()
