cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(recursive_hit_zoom VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
file(GLOB RECURSIVE_HIT_ZOOM_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(recursive_hit_zoom STATIC ${RECURSIVE_HIT_ZOOM_SOURCES})
add_library(chopfractal::recursive_hit_zoom ALIAS recursive_hit_zoom)
target_compile_features(recursive_hit_zoom PUBLIC cxx_std_17)
target_include_directories(recursive_hit_zoom PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
target_link_libraries(recursive_hit_zoom PUBLIC chopfractal::chop_contracts)
if(MSVC)
  target_compile_options(recursive_hit_zoom PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(recursive_hit_zoom PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB RECURSIVE_HIT_ZOOM_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_recursive_hit_zoom ${RECURSIVE_HIT_ZOOM_TESTS})
  target_link_libraries(test_recursive_hit_zoom PRIVATE recursive_hit_zoom)
  add_executable(example_recursive_hit_zoom examples/minimal.cpp)
  target_link_libraries(example_recursive_hit_zoom PRIVATE recursive_hit_zoom)
  if(MSVC)
    target_compile_options(test_recursive_hit_zoom PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_recursive_hit_zoom PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_recursive_hit_zoom PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_recursive_hit_zoom PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME recursive_hit_zoom.unit COMMAND test_recursive_hit_zoom)
  add_test(NAME recursive_hit_zoom.example COMMAND example_recursive_hit_zoom)
endif()
