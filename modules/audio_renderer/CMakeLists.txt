cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(audio_renderer VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
file(GLOB AUDIO_RENDERER_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(audio_renderer STATIC ${AUDIO_RENDERER_SOURCES})
add_library(chopfractal::audio_renderer ALIAS audio_renderer)
# Position independent so the module can be linked into a plugin (a shared module) in any project.
set_target_properties(audio_renderer PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(audio_renderer PUBLIC cxx_std_17)
target_include_directories(audio_renderer PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
target_link_libraries(audio_renderer PUBLIC chopfractal::chop_contracts)
if(MSVC)
  target_compile_options(audio_renderer PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(audio_renderer PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB AUDIO_RENDERER_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_audio_renderer ${AUDIO_RENDERER_TESTS})
  target_link_libraries(test_audio_renderer PRIVATE audio_renderer)
  add_executable(example_audio_renderer examples/minimal.cpp)
  target_link_libraries(example_audio_renderer PRIVATE audio_renderer)
  if(MSVC)
    target_compile_options(test_audio_renderer PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_audio_renderer PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_audio_renderer PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_audio_renderer PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME audio_renderer.unit COMMAND test_audio_renderer)
  add_test(NAME audio_renderer.example COMMAND example_audio_renderer)
endif()
