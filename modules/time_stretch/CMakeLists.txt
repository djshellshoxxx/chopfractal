cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(time_stretch VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
file(GLOB TIME_STRETCH_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(time_stretch STATIC ${TIME_STRETCH_SOURCES})
add_library(chopfractal::time_stretch ALIAS time_stretch)
# Position independent so the module can be linked into a plugin (a shared module) in any project.
set_target_properties(time_stretch PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(time_stretch PUBLIC cxx_std_17)
target_include_directories(time_stretch PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
target_link_libraries(time_stretch PUBLIC chopfractal::chop_contracts)
# Bit-identical results from the same binary: no fused multiply-add contraction in the correlation search.
if(MSVC)
  target_compile_options(time_stretch PRIVATE /fp:precise)
else()
  target_compile_options(time_stretch PRIVATE -ffp-contract=off)
endif()
if(MSVC)
  target_compile_options(time_stretch PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(time_stretch PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB TIME_STRETCH_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_time_stretch ${TIME_STRETCH_TESTS})
  target_link_libraries(test_time_stretch PRIVATE time_stretch)
  add_executable(example_time_stretch examples/minimal.cpp)
  target_link_libraries(example_time_stretch PRIVATE time_stretch)
  if(MSVC)
    target_compile_options(test_time_stretch PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_time_stretch PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_time_stretch PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_time_stretch PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME time_stretch.unit COMMAND test_time_stretch)
  add_test(NAME time_stretch.example COMMAND example_time_stretch)
endif()
