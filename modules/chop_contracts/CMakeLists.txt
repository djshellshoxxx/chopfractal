cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(chop_contracts VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()

file(GLOB CHOP_CONTRACTS_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(chop_contracts STATIC ${CHOP_CONTRACTS_SOURCES})
add_library(chopfractal::chop_contracts ALIAS chop_contracts)
# Position independent so the module can be linked into a plugin (a shared module) in any project.
set_target_properties(chop_contracts PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(chop_contracts PUBLIC cxx_std_17)
target_include_directories(chop_contracts PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")

if(MSVC)
  target_compile_options(chop_contracts PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(chop_contracts PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB CHOP_CONTRACTS_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_chop_contracts ${CHOP_CONTRACTS_TESTS})
  target_link_libraries(test_chop_contracts PRIVATE chop_contracts)
  add_executable(example_chop_contracts examples/minimal.cpp)
  target_link_libraries(example_chop_contracts PRIVATE chop_contracts)
  if(MSVC)
    target_compile_options(test_chop_contracts PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_chop_contracts PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_chop_contracts PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_chop_contracts PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME chop_contracts.unit COMMAND test_chop_contracts)
  add_test(NAME chop_contracts.example COMMAND example_chop_contracts)
endif()
