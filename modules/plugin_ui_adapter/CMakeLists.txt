cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(plugin_ui_adapter VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
if(NOT TARGET chopfractal::source_chop)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../source_chop" "${CMAKE_BINARY_DIR}/_chopfractal_source_chop")
endif()
if(NOT TARGET chopfractal::pattern_engine)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../pattern_engine" "${CMAKE_BINARY_DIR}/_chopfractal_pattern_engine")
endif()
if(NOT TARGET chopfractal::variation_history)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../variation_history" "${CMAKE_BINARY_DIR}/_chopfractal_variation_history")
endif()
file(GLOB PLUGIN_UI_ADAPTER_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(plugin_ui_adapter STATIC ${PLUGIN_UI_ADAPTER_SOURCES})
add_library(chopfractal::plugin_ui_adapter ALIAS plugin_ui_adapter)
target_compile_features(plugin_ui_adapter PUBLIC cxx_std_17)
target_include_directories(plugin_ui_adapter PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
target_link_libraries(plugin_ui_adapter PUBLIC chopfractal::chop_contracts chopfractal::source_chop chopfractal::pattern_engine chopfractal::variation_history)
if(MSVC)
  target_compile_options(plugin_ui_adapter PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(plugin_ui_adapter PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB PLUGIN_UI_ADAPTER_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_plugin_ui_adapter ${PLUGIN_UI_ADAPTER_TESTS})
  target_link_libraries(test_plugin_ui_adapter PRIVATE plugin_ui_adapter)
  add_executable(example_plugin_ui_adapter examples/minimal.cpp)
  target_link_libraries(example_plugin_ui_adapter PRIVATE plugin_ui_adapter)
  if(MSVC)
    target_compile_options(test_plugin_ui_adapter PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_plugin_ui_adapter PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_plugin_ui_adapter PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_plugin_ui_adapter PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME plugin_ui_adapter.unit COMMAND test_plugin_ui_adapter)
  add_test(NAME plugin_ui_adapter.example COMMAND example_plugin_ui_adapter)
endif()
