cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fractal_rhythm VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
file(GLOB FRACTAL_RHYTHM_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(fractal_rhythm STATIC ${FRACTAL_RHYTHM_SOURCES})
add_library(chopfractal::fractal_rhythm ALIAS fractal_rhythm)
# Position independent so the module can be linked into a plugin (a shared module) in any project.
set_target_properties(fractal_rhythm PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(fractal_rhythm PUBLIC cxx_std_17)
target_include_directories(fractal_rhythm PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
target_link_libraries(fractal_rhythm PUBLIC chopfractal::chop_contracts)
if(MSVC)
  target_compile_options(fractal_rhythm PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(fractal_rhythm PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB FRACTAL_RHYTHM_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_fractal_rhythm ${FRACTAL_RHYTHM_TESTS})
  target_link_libraries(test_fractal_rhythm PRIVATE fractal_rhythm)
  add_executable(example_fractal_rhythm examples/minimal.cpp)
  target_link_libraries(example_fractal_rhythm PRIVATE fractal_rhythm)
  if(MSVC)
    target_compile_options(test_fractal_rhythm PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_fractal_rhythm PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_fractal_rhythm PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_fractal_rhythm PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME fractal_rhythm.unit COMMAND test_fractal_rhythm)
  add_test(NAME fractal_rhythm.example COMMAND example_fractal_rhythm)
endif()
