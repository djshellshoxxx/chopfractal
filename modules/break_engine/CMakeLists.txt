cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(break_engine VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
file(GLOB BREAK_ENGINE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(break_engine STATIC ${BREAK_ENGINE_SOURCES})
add_library(chopfractal::break_engine ALIAS break_engine)
# Position independent so the module can be linked into a plugin (a shared module) in any project.
set_target_properties(break_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(break_engine PUBLIC cxx_std_17)
target_include_directories(break_engine PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
target_link_libraries(break_engine PUBLIC chopfractal::chop_contracts)
if(MSVC)
  target_compile_options(break_engine PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(break_engine PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB BREAK_ENGINE_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_break_engine ${BREAK_ENGINE_TESTS})
  target_link_libraries(test_break_engine PRIVATE break_engine)
  add_executable(example_break_engine examples/minimal.cpp)
  target_link_libraries(example_break_engine PRIVATE break_engine)
  if(MSVC)
    target_compile_options(test_break_engine PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_break_engine PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_break_engine PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_break_engine PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME break_engine.unit COMMAND test_break_engine)
  add_test(NAME break_engine.example COMMAND example_break_engine)
endif()
