cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(chop_roles_grammar VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET chopfractal::chop_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../chop_contracts" "${CMAKE_BINARY_DIR}/_chopfractal_chop_contracts")
endif()
file(GLOB CHOP_ROLES_GRAMMAR_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(chop_roles_grammar STATIC ${CHOP_ROLES_GRAMMAR_SOURCES})
add_library(chopfractal::chop_roles_grammar ALIAS chop_roles_grammar)
target_compile_features(chop_roles_grammar PUBLIC cxx_std_17)
target_include_directories(chop_roles_grammar PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
target_link_libraries(chop_roles_grammar PUBLIC chopfractal::chop_contracts)
if(MSVC)
  target_compile_options(chop_roles_grammar PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(chop_roles_grammar PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB CHOP_ROLES_GRAMMAR_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_chop_roles_grammar ${CHOP_ROLES_GRAMMAR_TESTS})
  target_link_libraries(test_chop_roles_grammar PRIVATE chop_roles_grammar)
  add_executable(example_chop_roles_grammar examples/minimal.cpp)
  target_link_libraries(example_chop_roles_grammar PRIVATE chop_roles_grammar)
  if(MSVC)
    target_compile_options(test_chop_roles_grammar PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_chop_roles_grammar PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_chop_roles_grammar PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_chop_roles_grammar PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME chop_roles_grammar.unit COMMAND test_chop_roles_grammar)
  add_test(NAME chop_roles_grammar.example COMMAND example_chop_roles_grammar)
endif()
