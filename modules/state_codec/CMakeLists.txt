cmake_minimum_required(VERSION 3.20)
# Standalone-capable: configure this folder directly, or add it from a parent project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(state_codec VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build module tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()

file(GLOB STATE_CODEC_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(state_codec STATIC ${STATE_CODEC_SOURCES})
add_library(chopfractal::state_codec ALIAS state_codec)
# Position independent so the module can be linked into a plugin (a shared module) in any project.
set_target_properties(state_codec PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(state_codec PUBLIC cxx_std_17)
target_include_directories(state_codec PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")

if(MSVC)
  target_compile_options(state_codec PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(state_codec PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB STATE_CODEC_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_state_codec ${STATE_CODEC_TESTS})
  target_link_libraries(test_state_codec PRIVATE state_codec)
  add_executable(example_state_codec examples/minimal.cpp)
  target_link_libraries(example_state_codec PRIVATE state_codec)
  if(MSVC)
    target_compile_options(test_state_codec PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    target_compile_options(example_state_codec PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
  else()
    target_compile_options(test_state_codec PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    target_compile_options(example_state_codec PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
  endif()
  add_test(NAME state_codec.unit COMMAND test_state_codec)
  add_test(NAME state_codec.example COMMAND example_state_codec)
endif()
