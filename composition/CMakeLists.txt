cmake_minimum_required(VERSION 3.20)
# The composition root: wires every portable module together. Headless (no JUCE, no GUI), so the whole
# load -> chop -> generate -> mutate/zoom -> branch -> save/reload -> playback flow is testable anywhere.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(chopfractal_composition VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  option(CHOPFRACTAL_BUILD_TESTS "Build tests" ON)
  option(CHOPFRACTAL_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
  if(CHOPFRACTAL_BUILD_TESTS)
    enable_testing()
  endif()
endif()

set(CHOPFRACTAL_COMPOSITION_DEPS chop_contracts state_codec variation_history source_chop pattern_engine
    recursive_hit_zoom chop_roles_grammar audio_renderer plugin_host_adapter)
foreach(dep ${CHOPFRACTAL_COMPOSITION_DEPS})
  if(NOT TARGET chopfractal::${dep})
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../modules/${dep}" "${CMAKE_BINARY_DIR}/_chopfractal_${dep}")
  endif()
endforeach()

file(GLOB COMPOSITION_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(composition STATIC ${COMPOSITION_SOURCES})
add_library(chopfractal::composition ALIAS composition)
target_compile_features(composition PUBLIC cxx_std_17)
target_include_directories(composition PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_LIST_DIR}/include>")
foreach(dep ${CHOPFRACTAL_COMPOSITION_DEPS})
  target_link_libraries(composition PUBLIC chopfractal::${dep})
endforeach()
if(MSVC)
  target_compile_options(composition PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
else()
  target_compile_options(composition PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
endif()

if(CHOPFRACTAL_BUILD_TESTS)
  file(GLOB COMPOSITION_TESTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/tests/*.cpp")
  add_executable(test_composition ${COMPOSITION_TESTS})
  target_link_libraries(test_composition PRIVATE composition)
  add_executable(example_composition examples/minimal.cpp)
  target_link_libraries(example_composition PRIVATE composition)
  foreach(t test_composition example_composition)
    if(MSVC)
      target_compile_options(${t} PRIVATE /W4 $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:/WX>)
    else()
      target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${CHOPFRACTAL_WARNINGS_AS_ERRORS}>:-Werror>)
    endif()
  endforeach()
  add_test(NAME composition.integration COMMAND test_composition)
  add_test(NAME composition.example COMMAND example_composition)
endif()
